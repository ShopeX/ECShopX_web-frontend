export class Cart {
  private _id: string
  private _items: Map<string, CartItemEntity>
  private _events: DomainEvent[]

  addItem(item: CartItemEntity): void {
    if (this._items.has(item.id)) {
      throw DomainError.alreadyExists('购物车商品', item.id)
    }
    if (!item.canBePurchased()) {
      throw DomainError.businessRuleViolation('商品不可购买')
    }
    this._items.set(item.id, item)
    this._events.push(new CartItemAddedEvent(...))  // ❌ 事件没用上
  }
}

export class CartPolicyService {
  mergeCartItem(cart: Cart, newItem: CartItemEntity): void {
    const existingItem = cart.items.find(i => i.productId === newItem.productId)
    if (existingItem) {
      existingItem.increaseQuantity(newItem.quantity)
    } else {
      cart.addItem(newItem)
    }
  }
}

// app/application/cart/CartApplicationService.ts (应用服务 - 254 行)
export class CartApplicationService {
  constructor(
    private readonly cartRepository: ICartRepository,
    private readonly cartPolicyService: CartPolicyService
  ) {}

  async addToCart(userId: string, dto: AddToCartDto): Promise<void> {
    // 1. 获取购物车
    let cart = await this.cartRepository.findById(userId)
    if (!cart) {
      cart = new Cart(userId, [])
    }

    // 2. 创建购物车项实体
    const item = new CartItemEntity({...dto})

    // 3. 使用领域服务合并商品
    this.cartPolicyService.mergeCartItem(cart, item)

    // 4. 保存购物车
    await this.cartRepository.save(cart)

    // 5. 清除领域事件
    cart.clearEvents()  // ❌ 只是清空，没有发布
  }
}

### ✅ 优化后（轻量 DDD，1 层调用）

```typescript
// app/composables/cart/useCartDomain.ts (简化后 ~150 行)

import { CartApiClient } from '~/infrastructure/http/clients/CartApiClient'
import { CartTransformer } from '~/infrastructure/transformers/cartTransformer'
import { MoneyValueObject } from '~/shared/value-objects/MoneyValueObject'
import { BusinessError } from '~/shared/errors/BusinessError'

export function useCartDomain() {
  const items = ref<ICartItem[]>([])
  const loading = ref(false)
  
  const cartApiClient = new CartApiClient()
  
  /**
   * 添加商品到购物车
   * 
   * 业务规则：
   * - 数量必须 > 0
   * - 数量不能超过库存
   * - 如果已存在则合并数量
   */
  const addToCart = async (product: Product, quantity: number) => {
    // ✅ 业务规则验证（直接在这里）
    if (quantity <= 0) {
      throw BusinessError.quantityInvalid(quantity)
    }
    if (quantity > product.stock) {
      throw BusinessError.stockInsufficient(product.name, product.stock)
    }
    
    // ✅ 业务逻辑：检查是否已存在
    const existingItem = items.value.find(i => i.productId === product.id)
    if (existingItem) {
      // 合并数量
      const newQuantity = existingItem.quantity + quantity
      if (newQuantity > product.stock) {
        throw BusinessError.stockInsufficient(
          product.name,
          product.stock - existingItem.quantity
        )
      }
      await updateQuantity(existingItem.id, newQuantity)
      return
    }
    
    // ✅ 调用 API（减少层级）
    loading.value = true
    try {
      await cartApiClient.addToCart({
        product_id: product.id,
        quantity,
        spec_id: product.specId,
      })
      
      await loadCart()
    } catch (err: any) {
      throw err
    } finally {
      loading.value = false
    }
  }
  
  /**
   * 计算选中商品总价
   * ✅ 使用值对象保证类型安全
   */
  const selectedTotal = computed(() => {
    return items.value
      .filter(item => item.selected)
      .reduce((sum, item) => {
        const price = MoneyValueObject.of(item.price)
        return sum + price.multiply(item.quantity).yuan
      }, 0)
  })
  
  return {
    items: readonly(items),
    loading: readonly(loading),
    selectedTotal,
    addToCart,
  }
}
```

```vue
<!-- app/pages/cart/index.vue (页面 - 不变) -->

    await addToCart(product, 1)
  } catch (err) {

  }
}
</script>
```

**优势总结**：
- ✅ 只需要修改 1-2 个文件
- ✅ 业务逻辑直观可见
- ✅ 仍然使用值对象保证类型安全
- ✅ 更容易调试和测试
- ✅ 新人容易理解（纯 TypeScript + Vue）

**代码量对比**：
- 优化前：~800 行（跨 5 个文件）

- **减少 75%** ✅

### ✅ 主题切换 Composable

```typescript
// app/composables/useTheme.ts

export function useTheme() {
  const currentThemeId = useCookie<string>('theme', {
    default: () => 'default',
  })
  
  const currentTheme = computed(() => {
    return THEMES[currentThemeId.value] || THEMES.default
  })
  
  /**
   * 切换主题
   */
  const setTheme = (themeId: string) => {
    currentThemeId.value = themeId
    updateThemeStyles(THEMES[themeId])
  }
  
  /**
   * 获取主题组件路径
   */
  const getThemeComponent = (category: string, name: string): string => {
    return currentTheme.value.components[category]?.[name]
  }
  
  return {
    currentTheme,
    setTheme,
    getThemeComponent,
  }
}
```
