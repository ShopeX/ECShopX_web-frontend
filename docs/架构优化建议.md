##### 优化后（推荐）

```typescript
// app/composables/cart/useCartDomain.ts
import { CartApiClient } from '~/infrastructure/http/clients/CartApiClient'
import { CartTransformer } from '~/infrastructure/transformers/cartTransformer'
import { MoneyValueObject } from '~/shared/value-objects/MoneyValueObject'

export function useCartDomain() {
  const items = ref<CartItem[]>([])
  const loading = ref(false)
  const error = ref<string | null>(null)

  // 直接调用 HTTP Client（减少层级）
  const cartApiClient = new CartApiClient()

  /**
   * 添加商品到购物车
   *
   * 业务规则：
   * - 数量必须 > 0
   * - 数量不能超过库存
   * - 如果已存在则合并数量
   */
  const addToCart = async (product: Product, quantity: number) => {
    // ✅ 业务规则验证直接在这里
    if (quantity <= 0) {
      throw new BusinessError('数量必须大于0')
    }
    if (quantity > product.stock) {
      throw new BusinessError(`库存不足，最多可购买 ${product.stock} 件`)
    }

    // ✅ 业务逻辑：检查是否已存在
    const existingItem = items.value.find((i) => i.productId === product.id)
    if (existingItem) {
      // 合并数量
      const newQuantity = existingItem.quantity + quantity
      if (newQuantity > product.stock) {
        throw new BusinessError(
          `购物车中已有 ${existingItem.quantity} 件，最多可再添加 ${product.stock - existingItem.quantity} 件`
        )
      }
      await updateQuantity(existingItem.id, newQuantity)
      return
    }

    // ✅ 调用 API
    loading.value = true
    try {
      await cartApiClient.addToCart({
        product_id: product.id,
        quantity,
        spec_id: product.specId,
      })

      // 刷新购物车
      await loadCart()
    } catch (err: any) {
      error.value = err.message
      throw err
    } finally {
      loading.value = false
    }
  }

  /**
   * 计算选中商品总价
   *
   * ✅ 使用值对象保证类型安全
   */
  const selectedTotal = computed(() => {
    return items.value
      .filter((item) => item.selected)
      .reduce((sum, item) => {
        const price = MoneyValueObject.of(item.price)
        return sum + price.multiply(item.quantity).yuan
      }, 0)
  })

  return {
    // 状态
    items: readonly(items),
    loading: readonly(loading),
    error: readonly(error),

    // 计算属性
    selectedTotal,

    // 方法
    loadCart,
    addToCart,
    updateQuantity,
    removeItem,
  }
}
```

**优势**：

- ✅ 减少 3 层调用为 1 层
- ✅ 业务逻辑直观可见
- ✅ 仍然使用值对象保证类型安全
- ✅ 更容易调试和测试
- ✅ 新人容易理解（纯 TypeScript + Vue）

#### 主题 Composable

```typescript
// app/composables/useTheme.ts

import { THEMES, type ThemeConfig } from '~/config/themes'

export function useTheme() {
  // 从 cookie 或 localStorage 读取当前主题
  const currentThemeId = useCookie<string>('theme', {
    default: () => 'default',
  })

  const currentTheme = computed<ThemeConfig>(() => {
    return THEMES[currentThemeId.value] || THEMES.default
  })

  /**
   * 切换主题
   */
  const setTheme = (themeId: string) => {
    if (!THEMES[themeId]) {
      throw new Error(`主题 ${themeId} 不存在`)
    }
    currentThemeId.value = themeId

    // 更新 CSS 变量
    updateThemeStyles(THEMES[themeId])
  }

  /**
   * 更新主题样式
   */
  const updateThemeStyles = (theme: ThemeConfig) => {
    if (process.client) {
      document.documentElement.style.setProperty('--primary-color', theme.styles.primaryColor)
      document.documentElement.style.setProperty('--border-radius', theme.styles.borderRadius)
      document.documentElement.style.setProperty('--spacing', theme.styles.spacing)
    }
  }

  /**
   * 获取主题组件路径
   */
  const getThemeComponent = (category: string, name: string): string => {
    const components = currentTheme.value.components as any
    return components[category]?.[name] || ''
  }

  // 初始化时应用主题样式
  onMounted(() => {
    updateThemeStyles(currentTheme.value)
  })

  return {
    currentTheme: readonly(currentTheme),
    currentThemeId: readonly(currentThemeId),
    availableThemes: Object.values(THEMES),
    setTheme,
    getThemeComponent,
  }
}
```

#### 标准 Transformer 模板

```typescript
// app/infrastructure/transformers/cartTransformer.ts

import type { ICartItem, ICartResponse } from '~/types/api/cart'
import { MoneyValueObject } from '~/shared/value-objects/MoneyValueObject'

/**
 * 购物车数据转换器
 *
 * 职责：
 * - API 响应 → 应用层数据模型
 * - 应用层数据模型 → UI 显示格式
 */
export class CartTransformer {
  /**
   * API 响应 → 应用层模型
   */
  static fromApiResponse(apiData: ICartResponse): ICartItem[] {
    return apiData.data.items.map((apiItem) => ({
      id: apiItem.id,
      productId: apiItem.product_id, // 字段映射
      productName: apiItem.product_name,
      productImage: apiItem.product_image,
      specId: apiItem.spec_id,
      specName: apiItem.spec_name,
      price: apiItem.price, // API 返回分
      marketPrice: apiItem.market_price,
      quantity: apiItem.quantity,
      stock: apiItem.stock,
      selected: apiItem.selected ?? true,
      available: apiItem.available ?? true,
    }))
  }

  /**
   * 应用层模型 → UI 显示格式
   */
  static toUIModel(item: ICartItem): ICartItemUI {
    const price = MoneyValueObject.of(item.price)
    const marketPrice = MoneyValueObject.of(item.marketPrice)
    const subtotal = price.multiply(item.quantity)

    return {
      ...item,
      // ✅ 使用值对象处理金额
      priceDisplay: price.display, // "¥99.90"
      marketPriceDisplay: marketPrice.display,
      subtotalDisplay: subtotal.display,

      // ✅ 业务计算
      hasDiscount: item.price < item.marketPrice,
      discountAmount: marketPrice.subtract(price).display,
      canBePurchased: item.available && item.stock >= item.quantity,
      stockWarning: item.stock < 10 ? `仅剩 ${item.stock} 件` : null,
    }
  }

  /**
   * 批量转换为 UI 格式
   */
  static toUIList(items: ICartItem[]): ICartItemUI[] {
    return items.map((item) => this.toUIModel(item))
  }
}
```

/**
 * 通用分页逻辑
 */
export function usePagination<T>(fetchFn: (page: number, pageSize: number) => Promise<T[]>) {
  const items = ref<T[]>([])
  const page = ref(1)
  const pageSize = ref(20)
  const total = ref(0)
  const loading = ref(false)

  const loadMore = async () => {
    if (loading.value) return

    loading.value = true
    try {
      const newItems = await fetchFn(page.value, pageSize.value)
      items.value.push(...newItems)
      page.value++
    } finally {
      loading.value = false
    }
  }

  return {
    items,
    page,
    pageSize,
    total,
    loading,
    loadMore,
  }
}

// 使用示例

#### 3. 日期和货币格式化

```typescript
// composables/useFormatters.ts
export function useFormatters() {
  const { locale } = useI18n()

  /**
   * 格式化货币
   */
  const formatCurrency = (amount: number, currency: string = 'CNY') => {
    return new Intl.NumberFormat(locale.value, {
      style: 'currency',
      currency,
    }).format(amount)
  }

  /**
   * 格式化日期
   */
  const formatDate = (date: Date) => {
    return new Intl.DateTimeFormat(locale.value, {
      year: 'numeric',
      month: 'long',
      day: 'numeric',
    }).format(date)
  }

  return {
    formatCurrency,
    formatDate,
  }
}

// 使用
const { formatCurrency } = useFormatters()
console.log(formatCurrency(99.9))
// zh-CN: ¥99.90
// en-US: $99.90
// ar-SA: 99.90 ر.س
```
